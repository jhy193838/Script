$('rec-add').onclick=()=>{uploadPublic('rec',$('rec-title').value,$('rec-code').value);$('rec-title').value='';$('rec-code').value='';};
function bindPublicFile(inputId,kind){$(inputId).onchange=e=>{[...e.target.files].forEach(file=>{const reader=new FileReader();reader.onload=ev=>uploadPublic(kind,file.name,String(ev.target.result??''));reader.readAsText(file,'UTF-8')});e.target.value='';};}
$('mine-file-btn').onclick=()=>$('mine-file').click();$('rec-file-btn').onclick=()=>$('rec-file').click();bindPublicFile('mine-file','mine');bindPublicFile('rec-file','rec');

// 채팅은 서버 없이 이 브라우저의 localStorage에만 저장됩니다.
const CHAT_KEY='script_hub_local_chat_v1';
let chatMessages=[];
try{const stored=JSON.parse(localStorage.getItem(CHAT_KEY)||'[]');if(Array.isArray(stored))chatMessages=stored;}catch(e){chatMessages=[];}
function saveChat(){try{localStorage.setItem(CHAT_KEY,JSON.stringify(chatMessages));}catch(e){toast('채팅 저장 공간이 부족합니다.');}}
function renderChat(){const box=$('chat-messages');box.innerHTML='';if(!chatMessages.length){box.innerHTML='<div class="chat-empty">아직 메시지가 없습니다.<br>첫 메시지를 남겨보세요.</div>';return;}chatMessages.forEach(m=>{const item=document.createElement('div');item.className='chat-message';const meta=document.createElement('div');meta.className='chat-meta';meta.textContent='나 · '+m.time;const body=document.createElement('div');body.className='chat-text';body.textContent=m.text;item.append(meta,body);box.appendChild(item);});box.scrollTop=box.scrollHeight;}
function sendChat(){const input=$('chat-input'),value=input.value.trim();if(!value)return;chatMessages.push({text:value,time:new Date().toLocaleString('ko-KR',{hour:'2-digit',minute:'2-digit'})});if(chatMessages.length>300)chatMessages=chatMessages.slice(-300);saveChat();renderChat();input.value='';input.focus();}
$('chat-send').onclick=sendChat;
$('chat-input').addEventListener('keydown',e=>{if(e.key==='Enter'&&!e.shiftKey){e.preventDefault();sendChat();}});
$('chat-clear').onclick=()=>{if(!chatMessages.length)return;if(confirm('이 브라우저에 저장된 채팅 기록을 모두 지울까요?')){chatMessages=[];saveChat();renderChat();}};
renderChat();

document.querySelectorAll('.tab').forEach(t=>t.onclick=()=>goPage(Number(t.dataset.page)));
